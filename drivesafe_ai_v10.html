Production Database & REST API Specification

PostgreSQL schema & REST endpoints powering DriveSafe AI automotive backend.

-- 1. SQL DATABASE SCHEMA (PostgreSQL / PostGIS)
CREATE EXTENSION IF NOT EXISTS postgis;

CREATE TABLE road_hazards (
    id SERIAL PRIMARY KEY,
    title VARCHAR(150) NOT NULL,
    hazard_type VARCHAR(50) NOT NULL,
    severity_score INT CHECK (severity_score BETWEEN 1 AND 10),
    address TEXT,
    coordinates GEOMETRY(Point, 4326),
    status VARCHAR(20) DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_hazards_spatial ON road_hazards USING GIST(coordinates);
// 2. PRODUCTION REST API ENDPOINTS
GET  /api/v1/hazards/radius?lat=23.55&lng=87.29&radius_km=5
POST /api/v1/hazards/report
POST /api/v1/ai/inspect-image
GET  /api/v1/routes/safe-path?origin=lat,lng&destination=lat,lng
